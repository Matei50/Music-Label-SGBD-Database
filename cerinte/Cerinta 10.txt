SET SERVEROUTPUT ON;

CREATE OR REPLACE TRIGGER ex10_lmd_comanda_vanzari
BEFORE DELETE ON vanzari
BEGIN
    IF TO_CHAR(SYSDATE, 'DY', 'NLS_DATE_LANGUAGE=ENGLISH') IN ('SAT','SUN') THEN
        RAISE_APPLICATION_ERROR(-20001,
        'Este interzisa stergerea vanzarilor in weekend!');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Stergerea este permisa.');
    END IF;
END;
/   

BEGIN
    DELETE FROM vanzari WHERE id_vanzari IN (1,2);
    ROLLBACK; -- ca sa nu modifici datele
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Trigger activat: ' || SQLERRM);
END;
/
